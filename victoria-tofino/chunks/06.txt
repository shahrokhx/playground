function buildRouteMap(p){
 const W=900,H=410,pad=36,pts=p.points;
 let minLat=Math.min(...pts.map(x=>x.lat)),maxLat=Math.max(...pts.map(x=>x.lat));
 let minLon=Math.min(...pts.map(x=>x.lon)),maxLon=Math.max(...pts.map(x=>x.lon));
 if(maxLat-minLat<.5){maxLat+=.25;minLat-=.25} if(maxLon-minLon<.8){maxLon+=.4;minLon-=.4}
 const latPad=(maxLat-minLat)*.12,lonPad=(maxLon-minLon)*.10; minLat-=latPad;maxLat+=latPad;minLon-=lonPad;maxLon+=lonPad;
 const xy=(lat,lon)=>({x:pad+(lon-minLon)/(maxLon-minLon)*(W-pad*2),y:H-pad-(lat-minLat)/(maxLat-minLat)*(H-pad*2)});
 const poly=(arr,cls)=>`<polygon class="${cls}" points="${arr.map(([lon,lat])=>{let q=xy(lat,lon);return `${q.x.toFixed(1)},${q.y.toFixed(1)}`}).join(' ')}"/>`;

 });
 const vic=xy(GEO.VIC.lat,GEO.VIC.lon), tof=xy(GEO.TOF.lat,GEO.TOF.lon);
 return `<div class="route-map-shell"><svg class="route-svg" viewBox="0 0 ${W} ${H}" role="img" aria-label="Schematic Vancouver Island route map">${grid}${poly(island,'land')}${poly(mainland,'land')}<polyline class="route-line" points="${route}"/>${pins}<text class="major-label" x="${vic.x+18}" y="${vic.y-18}">Victoria</text><text class="major-label" x="${tof.x+18}" y="${tof.y-18}">Tofino</text></svg><div class="map-legend"><span class="legend-route"></span><span>Trip sequence (schematic, not turn-by-turn road geometry)</span><span>• Tap a numbered stop to open its exact coordinates in Google Maps</span></div><div class="stop-strip">${chips}</div></div>`;
}
